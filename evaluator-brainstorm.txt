%1  ::= a :: bigint
%2  ::= b :: bigint
%3  ::= c :: boolean
%4  ::= if(%3, %1, %2)
%4a ::= %1 / %3
%4b ::= %2 / ~%3
%4  ::= union(%4a, %4b)
%5  ::= f(%1, %2)

* Memoization/buffering points. It can help make evaluation almost allocation-free in the steady state
  * any expression that is referenced multiple times
  * any expression that's evaluated conditionally and needs to be built up by evaluating multiple partial masks (e.g, then/else blocks in an if expression)
  * any expression that's evaluated once, but for the purpose of reusing buffers across multiple operator invocations
    * these are the lowest priority and can be dropped under memory pressure

* In the expression IR, a mask is any boolean column
   * T is the special mask that is always true


%1  ::= a / T :: bigint
%2  ::= b / T :: bigint
%3  ::= c / T :: boolean
%4a ::= %1 / %3
%4b ::= %2 / ~%3
%4  ::= union(%4a, %4b) / T
%5  ::= f(%1, %2) / T

%3  = d / T

%6  = %2 / %0
%7  = union(%4, %5) / T
%8  = union(%6, %5) / T
%9  = +(%7, %8, %1, %2) / T


====

#if(a, b, c)

%0 = a / T
%1 = EQ(%0, T) / T
%2 = b / %1
%3 = c / ~%1
$r = union(%2, %3) / T

====
#coalesce(a, b, c)

%0  = a / T
%1r = %0 / ~%0.nulls

%2  = b / %0.nulls
%3  = IS_NULL(%2) / %0.nulls
%3r = %2 / ~%3

%4  = c / %3

%r = merge(%1r, %3r, %4) / T

====

#try(f : () -> ())

%0 = f() / T
%1 = IS_ERROR(%0) / T
%2 = NULL / %1
%3 = %0 / ~%1
%r = union(%2, %3) / T


====

#nullif(a, b)

%0 = a / T
%1 = b / T
%2 = EQ(%0, %1) / T
%3 = NULL / %2
%4 = %0 / ~%2
%r = union(%3, %4) / T


====

#and(a, b)

# T AND T = T
# T AND F = F
# F AND T = F
# F AND F = F
# T AND NULL = NULL
# F AND NULL = F
# NULL AND T = NULL
# NULL AND F = F
# NULL AND NULL = NULL

%0 = a / T
%1 = IDENTICAL(%0, T) / T
%2 = IDENTICAL(%0, F) / T

%3 = b / %1
%4 = IDENTICAL(%3, T) / %1

%4 = NULL / ~%1 //
%5 = %2 / ~%3
%r = union(%4, %5) / T
